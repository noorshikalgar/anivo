  exit title fade dur=300ms
  enter producer rise
  enter producer->m1 draw-on after=prev
  enter m1, m2, m3, m4 slide-left stagger=180ms after=prev
  enter queue-label fade after=prev

  enter broker pop
  enter m4->broker draw-on after=prev
  enter w1, w2, w3 rise stagger=150ms after=prev
  enter broker->w1, broker->w2, broker->w3 draw-on after=prev
step "Consume" caption="Each message goes to one worker."
  flow broker->w* round-robin rate=2
  exit m4, m3 fade stagger=600ms +600ms
step "A worker fails" caption="A worker is down: the queue keeps the messages until a worker is free."
  morph w2 fill=danger-soft stroke=danger label="Worker 2 down"
  circle w2 color=danger after=prev
  flow broker->w* round-robin skip=broker->w2 rate=2

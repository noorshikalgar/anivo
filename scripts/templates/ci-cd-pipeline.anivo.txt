scene "CI/CD pipeline" style=clean theme=studio-dark size=1920x1080 fps=60 background=gradient

text title "From a commit to production" at=160,430 size=1600x200 font-size=88 text-align=center
row stages at=110,430 spacing=110
  box commit "Commit" size=300x150 look=card font-size=40
  box build "Build" size=300x150 look=card font-size=40
  box test "Test" size=300x150 look=card font-size=40
  box deploy "Deploy" size=300x150 look=card font-size=40
text tests "0 tests passed" at=660,680 size=600x100 font-size=64 text-align=center
icon prod lucide:rocket "Production" at=1660,140 size=180x180 fill=accent font-size=36

arrow commit -> build
arrow build -> test
arrow test -> deploy
arrow deploy -> prod

step "Title" hold=600ms
  enter title cascade unit=word dur=1300ms easing=linear
step "Push a commit" caption="You push a commit. The pipeline starts on its own." transition=blur
  exit title fade dur=300ms
  enter commit, build, test, deploy rise stagger=150ms
  enter commit->build, build->test, test->deploy draw-on stagger=150ms after=prev
step "Build" caption="The build server compiles the code and makes an image."
  flow commit->build once icon=lucide:git-commit-horizontal size=40
  pulse build after=prev
  morph build fill=success-soft stroke=success after=prev
step "Test" caption="Every test runs. One failure stops the pipeline."
  flow build->test once icon=lucide:package size=40
  enter tests fade after=prev
  count tests from=0 to=1280 format="{n} tests passed" style=thousands roll dur=2000ms after=prev
  morph test fill=success-soft stroke=success after=prev
step "Deploy" caption="The image goes to production, a few servers at a time."
  flow test->deploy once icon=lucide:package size=40
  morph deploy fill=success-soft stroke=success after=prev
  enter prod pop after=prev
  enter deploy->prod draw-on after=prev
  float prod after=prev
